function showTab(name){

  ['historial','stats'].forEach(t=>{

    document.getElementById(`tab-${t}`).style.display=t===name?'block':'none';

  });

  if(name==='stats') renderStats();

}


// ── INIT ──────────────────────────────────────────────────────────────────────

async function init(){

  // Obtener nombre del cliente desde la URL

  const path=window.location.pathname;

  const encoded=path.split('/cliente/')[1];

  if(!encoded){ mostrarError('Cliente no especificado'); return; }

  nombreCliente=decodeURIComponent(encoded);


  if(nombreCliente==='__nuevo__'){

    mostrarFormularioNuevo();

    return;

  }


  document.title=`${nombreCliente} — La Guarida`;

  document.getElementById('page-title').textContent=nombreCliente;


  // Iniciales para avatar

  const iniciales=nombreCliente.split(' ').map(w=>w[0]).join('').slice(0,2).toUpperCase();

  document.getElementById('avatar-initials').textContent=iniciales;


  try{

    // Cargar datos en paralelo

    const [fichas,pedidosTodos,clientes]=await Promise.all([

      api('/api/clientes_fichas'),

      api('/api/pedidos'),

      api('/api/clientes')

    ]);


    todosClientes=clientes;

    fichaData=fichas[nombreCliente]||{};

    pedidosCliente=pedidosTodos.filter(p=>p.cliente===nombreCliente)

      .sort((a,b)=>b.fecha.localeCompare(a.fecha));


    renderFicha();

    renderKPIs();

    renderHistorial();

    renderProductosFrecuentes();

    poblarUnificar(clientes);


  }catch(e){

    console.error(e);

    mostrarError('Error al cargar los datos del cliente.');

  }

}

  document.getElementById('page-title').textContent='Error';

  document.getElementById('page-content').innerHTML=`<div class="card"><p style="color:var(--red);padding:20px">${msg}</p></div>`;

}

  mostrarEdicion();

}


// ── FICHA ─────────────────────────────────────────────────────────────────────

function renderFicha(){

  const f=fichaData;

  document.getElementById('ficha-nombre-display').textContent=nombreCliente;


  // Estado basado en último pedido

  const hoy=new Date();

  const ult=pedidosCliente[0]?.fecha;

  const dias=ult?Math.floor((hoy-new Date(ult))/(1000*60*60*24)):999;

  let estadoTxt,estadoCls;

  if(dias<=30){estadoTxt='🟢 Cliente activo';estadoCls='estado-activo';}

  else if(dias<=90){estadoTxt='🟡 Compra poco frecuente';estadoCls='estado-tibio';}

  else{estadoTxt='🔴 Sin compras recientes';estadoCls='estado-contactar';}

  document.getElementById('ficha-estado-display').innerHTML=`<span class="${estadoCls}">${estadoTxt}</span>`;


  document.getElementById('view-dni').textContent=f.dni_cuit||'—';

  document.getElementById('view-tel').textContent=f.telefono||'—';

  document.getElementById('view-dir').textContent=f.direccion||'—';

  document.getElementById('view-ciudad').textContent=f.ciudad||'—';

  document.getElementById('view-prov').textContent=f.provincia||'—';

  document.getElementById('view-cp').textContent=f.cp||'—';

  document.getElementById('view-notas').textContent=f.notas||'Sin notas';

}

async function guardarFicha(){

  const nombre=nombreCliente==='__nuevo__'

    ? prompt('Nombre del cliente:')?.trim()

    : nombreCliente;

  if(!nombre){toast('El nombre es obligatorio',true);return;}


  const data={

    dni_cuit:document.getElementById('edit-dni').value,

    telefono:document.getElementById('edit-tel').value,

    direccion:document.getElementById('edit-dir').value,

    ciudad:document.getElementById('edit-ciudad').value,

    provincia:document.getElementById('edit-prov').value,

    cp:document.getElementById('edit-cp').value,

    notas:document.getElementById('edit-notas').value,

  };


  try{

    await api(`/api/clientes_fichas/${encodeURIComponent(nombre)}`,{method:'POST',body:JSON.stringify(data)});

    fichaData={...data};

    if(nombreCliente==='__nuevo__'){

      nombreCliente=nombre;

      window.history.replaceState(null,'',`/cliente/${encodeURIComponent(nombre)}`);

      document.getElementById('page-title').textContent=nombre;

      document.getElementById('avatar-initials').textContent=nombre.split(' ').map(w=>w[0]).join('').slice(0,2).toUpperCase();

      document.getElementById('ficha-nombre-display').textContent=nombre;

    }

    cancelarEdicion();

    renderFicha();

    toast('✅ Ficha guardada');

  }catch(e){toast('Error al guardar',true);}

}


// ── KPIs ──────────────────────────────────────────────────────────────────────

function renderKPIs(){

  const total=pedidosCliente.reduce((s,p)=>s+(p.total||0),0);

  const n=pedidosCliente.length;

  const ticket=n?Math.round(total/n):0;


  // Frecuencia: días promedio entre pedidos

  let frecuencia='—';

  if(n>=2){

    const fechas=pedidosCliente.map(p=>new Date(p.fecha)).sort((a,b)=>a-b);

    const diffs=[];

    for(let i=1;i<fechas.length;i++) diffs.push((fechas[i]-fechas[i-1])/(1000*60*60*24));

    const prom=Math.round(diffs.reduce((s,d)=>s+d,0)/diffs.length);

    frecuencia=`cada ${prom}d`;

  }


  const ult=pedidosCliente[0]?.fecha||'—';

  const hoy=new Date();

  const dias=ult!=='—'?Math.floor((hoy-new Date(ult))/(1000*60*60*24)):999;

  let estadoHTML;

  if(dias<=30) estadoHTML='<span class="estado-activo">🟢 Activo</span>';

  else if(dias<=90) estadoHTML='<span class="estado-tibio">🟡 Tibio</span>';

  else estadoHTML='<span class="estado-contactar">🔴 Contactar</span>';


  document.getElementById('kpi-total').textContent=fmt(total);

  document.getElementById('kpi-pedidos').textContent=n;

  document.getElementById('kpi-ticket').textContent=fmt(ticket);

  document.getElementById('kpi-frecuencia').textContent=frecuencia;

  document.getElementById('kpi-ultimo').textContent=ult;

  document.getElementById('kpi-estado').innerHTML=estadoHTML;


  // Subtitle

  document.getElementById('page-subtitle').textContent=

    `${n} pedido${n!==1?'s':''} · ${fmt(total)} acumulado`;

}

// ── STATS ─────────────────────────────────────────────────────────────────────

function renderStats(){

  if(!pedidosCliente.length){

    document.getElementById('stats-content').innerHTML='<div class="empty">Sin datos suficientes</div>';

    return;

  }


  // Ventas por mes

  const porMes={};

  pedidosCliente.forEach(p=>{

    const mes=p.fecha.slice(0,7);

    if(!porMes[mes]) porMes[mes]=0;

    porMes[mes]+=p.total||0;

  });

  const meses=Object.keys(porMes).sort();

  const maxMes=Math.max(...Object.values(porMes),1);


  // Productos

  const porProd={};

  pedidosCliente.forEach(p=>{

    (p.items||[]).forEach(i=>{

      if(!porProd[i.producto]) porProd[i.producto]={unidades:0,total:0};

      porProd[i.producto].unidades+=i.cantidad||0;

      porProd[i.producto].total+=i.subtotal||0;

    });

  });

  const prodOrdenado=Object.entries(porProd).sort((a,b)=>b[1].total-a[1].total);


  // Año actual vs anterior

  const anioActual=new Date().getFullYear();

  const totalActual=pedidosCliente.filter(p=>p.fecha.startsWith(anioActual)).reduce((s,p)=>s+(p.total||0),0);

  const totalAnterior=pedidosCliente.filter(p=>p.fecha.startsWith(anioActual-1)).reduce((s,p)=>s+(p.total||0),0);


  document.getElementById('stats-content').innerHTML=`

    <div style="display:grid;grid-template-columns:1fr 1fr;gap:12px;margin-bottom:16px">

      <div style="background:var(--bg);border:1px solid var(--border);border-radius:8px;padding:14px;text-align:center">

        <div style="font-size:10px;color:var(--muted);text-transform:uppercase;margin-bottom:4px">Ventas ${anioActual}</div>

        <div style="font-size:22px;font-weight:800;color:var(--accent)">${fmt(totalActual)}</div>

      </div>

      <div style="background:var(--bg);border:1px solid var(--border);border-radius:8px;padding:14px;text-align:center">

        <div style="font-size:10px;color:var(--muted);text-transform:uppercase;margin-bottom:4px">Ventas ${anioActual-1}</div>

        <div style="font-size:22px;font-weight:800;color:var(--muted)">${fmt(totalAnterior)}</div>

      </div>

    </div>


    <div style="font-size:11px;font-weight:700;color:var(--muted);text-transform:uppercase;letter-spacing:.5px;margin-bottom:10px">Compras por mes</div>

    ${meses.map(m=>{

      const [y,mo]=m.split('-');

      const label=new Date(y,parseInt(mo)-1).toLocaleString('es-AR',{month:'short',year:'2-digit'});

      const pct=Math.round(porMes[m]/maxMes*100);

      return `<div class="freq-bar">

        <span class="freq-label">${label}</span>

        <div class="freq-track">

          <div class="freq-fill" style="width:${pct}%">

            <span class="freq-val">${pct>15?fmt(porMes[m]):''}</span>

          </div>

        </div>

        <span style="font-size:11px;color:var(--accent);font-weight:700;min-width:80px;text-align:right">${fmt(porMes[m])}</span>

      </div>`;

    }).join('')}


    <div style="font-size:11px;font-weight:700;color:var(--muted);text-transform:uppercase;letter-spacing:.5px;margin:16px 0 10px">Productos comprados</div>

    <table>

      <thead><tr>

        <th>Producto</th>

        <th class="tc">Unidades</th>

        <th class="tr">Total $</th>

      </tr></thead>

      <tbody>

        ${prodOrdenado.map(([nombre,d])=>`<tr>

          <td>${nombre}</td>

          <td class="tc">${d.unidades}</td>

          <td class="tr" style="color:var(--accent);font-weight:700">${fmt(d.total)}</td>

        </tr>`).join('')}

      </tbody>

    </table>`;

}


// ── PRODUCTOS FRECUENTES ──────────────────────────────────────────────────────

function renderProductosFrecuentes(){

  const porProd={};

  pedidosCliente.forEach(p=>{

    (p.items||[]).forEach(i=>{

      if(!porProd[i.producto]) porProd[i.producto]={unidades:0,veces:0};

      porProd[i.producto].unidades+=i.cantidad||0;

      porProd[i.producto].veces++;

    });

  });

  const lista=Object.entries(porProd).sort((a,b)=>b[1].unidades-a[1].unidades).slice(0,5);

  const maxU=lista[0]?.[1]?.unidades||1;


  if(!lista.length){

    document.getElementById('productos-frecuentes').innerHTML='<div class="empty" style="padding:16px">Sin productos registrados</div>';

    return;

  }


  document.getElementById('productos-frecuentes').innerHTML=lista.map(([nombre,d])=>{

    const pct=Math.round(d.unidades/maxU*100);

    return `<div class="freq-bar" style="margin-bottom:10px">

      <div style="flex:1">

        <div style="display:flex;justify-content:space-between;font-size:11px;margin-bottom:3px">

          <span style="font-weight:600">${nombre}</span>

          <span style="color:var(--accent);font-weight:700">${d.unidades} u</span>

        </div>

        <div class="freq-track">

          <div class="freq-fill" style="width:${pct}%;background:var(--blue)">

            <span class="freq-val">${d.veces} pedido${d.veces!==1?'s':''}</span>

          </div>

        </div>

      </div>

    </div>`;

  }).join('');

}


// ── UNIFICAR ──────────────────────────────────────────────────────────────────

function poblarUnificar(clientes){

  const sel=document.getElementById('unif-select');

  sel.innerHTML='<option value="">— Elegí el duplicado —</option>'+

    clientes.filter(c=>c.cliente!==nombreCliente)

      .map(c=>`<option value="${c.cliente}">${c.cliente} (${c.num_pedidos} pedidos · ${fmt(c.total_acumulado)})</option>`)

      .join('');

  sel.addEventListener('change',function(){

    const dup=this.value;

    const prev=document.getElementById('unif-preview');

    if(!dup){prev.style.display='none';return;}

    prev.style.display='block';

    prev.textContent=`Todos los pedidos de "${dup}" se moverán a "${nombreCliente}" y el duplicado se eliminará.`;

  });

}


function abrirUnificar(){

  const panel=document.getElementById('unificar-panel');

  panel.style.display=panel.style.display==='none'?'block':'none';

}


async function ejecutarUnificar(){

  const duplicado=document.getElementById('unif-select').value;

  if(!duplicado){toast('Elegí el duplicado',true);return;}

  if(!confirm(`¿Unificar "${duplicado}" en "${nombreCliente}"? No se puede deshacer.`)) return;

  try{

    const pedidosTodos=await api('/api/pedidos');

    const pedsDup=pedidosTodos.filter(p=>p.cliente===duplicado);

    for(const p of pedsDup){

      await api(`/api/pedidos/${p.id}`,{method:'PUT',body:JSON.stringify({...p,cliente:nombreCliente})});

    }

    try{await api(`/api/clientes_fichas/${encodeURIComponent(duplicado)}`,{method:'DELETE'});}catch(e){}

    toast(`✅ "${duplicado}" unificado`);

    document.getElementById('unificar-panel').style.display='none';

    // Recargar

    const pedidosTodos2=await api('/api/pedidos');

    pedidosCliente=pedidosTodos2.filter(p=>p.cliente===nombreCliente).sort((a,b)=>b.fecha.localeCompare(a.fecha));

    renderKPIs();

    renderHistorial();

    renderProductosFrecuentes();

  }catch(e){toast('Error al unificar',true);}

}
